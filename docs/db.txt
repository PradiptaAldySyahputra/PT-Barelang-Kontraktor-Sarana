// ERD KONSEPTUAL SISTEM INFORMASI SPK & KONTROL KEUANGAN
// SPK adalah entitas inti. Tidak ada entitas PROYEK.

Table pengguna {
  id int [pk, increment]
  nama varchar(150) [not null]
  email varchar(150) [not null, unique]
  peran varchar(50) [not null]
}

Table mitra {
  id int [pk, increment]
  nama varchar(150) [not null]
  kategori varchar(50) [not null]
  kontak varchar(100)
}

Table spk {
  id int [pk, increment]
  nomor_spk varchar(100) [not null, unique]
  tanggal_spk date [not null]
  nama_pekerjaan varchar(200) [not null]
  lokasi varchar(200)
  nilai_spk decimal(18,2) [not null]
  pic varchar(150)
  status_spk varchar(50)
  status_tagihan varchar(50)
  keterangan text
  mitra_id int
  dibuat_oleh int [not null]
}

Table uang_masuk {
  id int [pk, increment]
  nomor_transaksi varchar(100) [not null, unique]
  tanggal date [not null]
  jumlah decimal(18,2) [not null]
  sumber varchar(50) [not null, note: 'SPK / Luar SPK']
  spk_id int [note: 'Boleh kosong jika berasal dari luar SPK']
  mitra_id int
  metode_pembayaran varchar(50)
  status varchar(50)
  dicatat_oleh int [not null]
}

Table uang_keluar {
  id int [pk, increment]
  nomor_transaksi varchar(100) [not null, unique]
  tanggal date [not null]
  jumlah decimal(18,2) [not null]
  kategori varchar(100) [not null]
  spk_id int [note: 'Boleh kosong untuk pengeluaran umum']
  mitra_id int
  metode_pembayaran varchar(50)
  status varchar(50)
  dicatat_oleh int [not null]
  disetujui_oleh int
  tanggal_persetujuan datetime
}

Table bukti {
  id int [pk, increment]
  nama_file varchar(255) [not null]
  jenis_bukti varchar(100)
  lokasi_file varchar(500) [not null]
  diunggah_oleh int [not null]
  terkait_dengan varchar(20) [not null, note: 'SPK / UANG_MASUK / UANG_KELUAR']
  id_data_terkait int [not null]
}

Ref: pengguna.id < spk.dibuat_oleh
Ref: pengguna.id < uang_masuk.dicatat_oleh
Ref: pengguna.id < uang_keluar.dicatat_oleh
Ref: pengguna.id < uang_keluar.disetujui_oleh
Ref: pengguna.id < bukti.diunggah_oleh
Ref: mitra.id < spk.mitra_id
Ref: mitra.id < uang_masuk.mitra_id
Ref: mitra.id < uang_keluar.mitra_id
Ref: spk.id ?< uang_masuk.spk_id
Ref: spk.id ?< uang_keluar.spk_id
